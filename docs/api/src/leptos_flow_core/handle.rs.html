1//! Handle system for precise node connections
2//!
3//! Handles are connection points on nodes that allow for precise edge attachment.
4//! They support input/output semantics, type validation, and positioning.
5

12/// Handle identifier
13#[derive(Debug, Clone, PartialEq, Eq, Hash)]
14#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
15pub struct HandleId(pub String);
16
17impl HandleId {
18    /// Create a new handle ID
19    pub fn new(id: impl Into<String>) -> Self {
20        Self(id.into())
21    }
22
23    /// Get handle ID as string
24    pub fn as_str(&self) -> &str {
25        &self.0
26    }
27}
28
29impl From<&str> for HandleId {
30    fn from(id: &str) -> Self {
31        Self(id.to_string())
32    }
33}
34
35impl From<String> for HandleId {
36    fn from(id: String) -> Self {
37        Self(id)
38    }
39}

41/// Handle type indicating connection direction
42#[derive(Debug, Clone, PartialEq)]
43#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
44pub enum HandleType {
45    Source,
46    Target,
47}
48
49impl HandleType {
50    /// Check if this is a source handle
51    pub fn is_source(&self) -> bool {
52        matches!(self, HandleType::Source)
53    }
54
55    /// Check if this is a target handle
56    pub fn is_target(&self) -> bool {
57        matches!(self, HandleType::Target)
58    }
59}

61/// Handle position relative to node
62#[derive(Debug, Clone, PartialEq)]
63#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
64pub enum HandlePosition {
65    Top,
66    Right,
67    Bottom,
68    Left,
69    Custom(Position),
70}
71
72impl HandlePosition {
73    /// Convert handle position to absolute position relative to node
74    pub fn to_position(&self) -> Position {
75        match self {
76            HandlePosition::Top => Position::new(0.0, -10.0),
77            HandlePosition::Right => Position::new(80.0, 30.0),
78            HandlePosition::Bottom => Position::new(40.0, 60.0),
79            HandlePosition::Left => Position::new(0.0, 30.0),
80            HandlePosition::Custom(pos) => *pos,
81        }
82    }
83}

85/// Connection handle on a node
86#[derive(Debug, Clone, PartialEq)]
87#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
88pub struct Handle {
89    pub id: HandleId,
90    pub handle_type: HandleType,
91    pub position: HandlePosition,
92    pub connection_limit: Option<usize>,
93    pub valid_connection_types: Option<Vec<String>>,
94    pub style: Option<String>,
95}
96
97impl Handle {
98    /// Create a new handle
99    pub fn new(
100        id: impl Into<HandleId>,
101        handle_type: HandleType,
102        position: HandlePosition,
103    ) -> Self {
104        Self {
105            id: id.into(),
106            handle_type,
107            position,
108            connection_limit: None,
109            valid_connection_types: None,
110            style: None,
111        }
112    }
113
114    /// Create a source handle
115    pub fn source(id: impl Into<HandleId>, position: HandlePosition) -> Self {
116        Self::new(id, HandleType::Source, position)
117    }
118
119    /// Create a target handle
120    pub fn target(id: impl Into<HandleId>, position: HandlePosition) -> Self {
121        Self::new(id, HandleType::Target, position)
122    }
123
124    /// Set connection limit
125    pub fn with_connection_limit(mut self, limit: usize) -> Self {
126        self.connection_limit = Some(limit);
127        self
128    }
129
130    /// Set valid connection types
131    pub fn with_connection_types(mut self, types: Vec<String>) -> Self {
132        self.valid_connection_types = Some(types);
133        self
134    }
135
136    /// Set style
137    pub fn with_style(mut self, style: impl Into<String>) -> Self {
138        self.style = Some(style.into());
139        self
140    }
141
142    /// Calculate absolute position of handle given node position and size
143    pub fn absolute_position(&self, node_pos: Position, node_size: Size) -> Position {
144        match &self.position {
145            HandlePosition::Top => Position::new(
146                node_pos.x + node_size.width / 2.0,
147                node_pos.y,
148            ),
149            HandlePosition::Right => Position::new(
150                node_pos.x + node_size.width,
151                node_pos.y + node_size.height / 2.0,
152            ),
153            HandlePosition::Bottom => Position::new(
154                node_pos.x + node_size.width / 2.0,
155                node_pos.y + node_size.height,
156            ),
157            HandlePosition::Left => Position::new(
158                node_pos.x,
159                node_pos.y + node_size.height / 2.0,
160            ),
161            HandlePosition::Custom(pos) => Position::new(
162                node_pos.x + pos.x,
163                node_pos.y + pos.y,
164            ),
165        }
166    }
167
168    /// Check if a point is within the handle's bounds
169    pub fn contains_point(&self, point: Position, node_pos: Position, node_size: Size, handle_size: f64) -> bool {
170        let handle_pos = self.absolute_position(node_pos, node_size);
171        let half_size = handle_size / 2.0;
172
173        point.x >= handle_pos.x - half_size &&
174        point.x <= handle_pos.x + half_size &&
175        point.y >= handle_pos.y - half_size &&
176        point.y <= handle_pos.y + half_size
177    }
178
179    /// Check if this handle can connect to another handle
180    pub fn can_connect_to(&self, other: &Handle) -> bool {
181        // Can't connect to same type (source to source, target to target)
182        if self.handle_type == other.handle_type {
183            return false;
184        }
185
186        // Check connection type compatibility
187        if let (Some(self_types), Some(other_types)) = (&self.valid_connection_types, &other.valid_connection_types) {
188            // At least one matching type required
189            return self_types.iter().any(|t| other_types.contains(t));
190        }
191
192        true // No type restrictions
193    }
194}

196/// Handle manager for a node
197#[derive(Debug, Clone, PartialEq)]
198#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
199pub struct HandleManager {
200    handles: Vec<Handle>,
201    node_id: NodeId,
202}
203
204impl Default for HandleManager {
205    fn default() -> Self {
206        Self {
207            handles: Vec::new(),
208            node_id: "default".into(),
209        }
210    }
211}
212
213impl HandleManager {
214    /// Create a new handle manager
215    pub fn new(node_id: NodeId) -> Self {
216        Self {
217            handles: Vec::new(),
218            node_id,
219        }
220    }
221
222    /// Add a handle
223    pub fn add_handle(&mut self, handle: Handle) -> Result<()> {
224        // Check for duplicate handle IDs
225        if self.handles.iter().any(|h| h.id == handle.id) {
226            return Err(FlowError::invalid_operation(
227                &format!("Handle '{}' already exists", handle.id.as_str())
228            ));
229        }
230
231        self.handles.push(handle);
232        Ok(())
233    }
234
235    /// Remove a handle
236    pub fn remove_handle(&mut self, handle_id: &HandleId) -> Result<Handle> {
237        let index = self.handles.iter().position(|h| &h.id == handle_id)
238            .ok_or_else(|| FlowError::invalid_operation(
239                &format!("Handle '{}' not found", handle_id.as_str())
240            ))?;
241
242        Ok(self.handles.remove(index))
243    }
244
245    /// Get a handle by ID
246    pub fn get_handle(&self, handle_id: &HandleId) -> Option<&Handle> {
247        self.handles.iter().find(|h| &h.id == handle_id)
248    }
249
250    /// Get all handles
251    pub fn handles(&self) -> &[Handle] {
252        &self.handles
253    }
254
255    /// Find handle at position
256    pub fn handle_at_position(&self, point: Position, node_pos: Position, node_size: Size, handle_size: f64) -> Option<&Handle> {
257        self.handles.iter().find(|handle| {
258            handle.contains_point(point, node_pos, node_size, handle_size)
259        })
260    }
261
262    /// Get source handles
263    pub fn source_handles(&self) -> impl Iterator<Item = &Handle> {
264        self.handles.iter().filter(|h| h.handle_type == HandleType::Source)
265    }
266
267    /// Get target handles
268    pub fn target_handles(&self) -> impl Iterator<Item = &Handle> {
269        self.handles.iter().filter(|h| h.handle_type == HandleType::Target)
270    }
271
272    /// Count connections for a handle
273    ///
274    /// Note: This method requires graph integration to work properly.
275    /// Use `Graph::get_handle_connections()` directly for accurate counting.
276    pub fn connection_count(&self, _handle_id: &HandleId) -> usize {
277        // This method is deprecated in favor of Graph-level connection counting
278        // because it needs access to the graph's edge collection to count connections accurately.
279        // Use graph.get_handle_connections(node_id, handle_id).len() instead.
280        0
281    }
282
283    /// Check if handle can accept new connections
284    ///
285    /// Note: This method requires graph integration to work properly.
286    /// Use `Graph::can_handle_accept_connection()` for accurate validation.
287    pub fn can_accept_connection(&self, handle_id: &HandleId) -> bool {
288        if let Some(handle) = self.get_handle(handle_id) {
289            if let Some(_limit) = handle.connection_limit {
290                // This method is deprecated in favor of Graph-level validation
291                // because it needs access to the graph's edge collection.
292                // Use graph.can_handle_accept_connection(node_id, handle_id) instead.
293                return true; // Conservative default
294            }
295        }
296        true
297    }
