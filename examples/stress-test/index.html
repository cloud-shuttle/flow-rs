Efficient Graph Generation

fn generate_large_graph(node_count: usize) -> Graph<(), ()> {
    let mut graph = Graph::new();

    // Calculate optimal grid layout
    let cols = (node_count as f64).sqrt().ceil() as usize;
    let spacing_x = 1200.0 / (cols as f64);

    // Create nodes with spatial distribution
    for i in 0..node_count {
        let x = (i % cols) as f64 * spacing_x;
        let y = (i / cols) as f64 * spacing_y;

        // Add controlled randomness for natural layout
        let x = x + (random() - 0.5) * spacing_x * 0.3;
        let y = y + (random() - 0.5) * spacing_y * 0.3;

        graph.add_node(Node::simple(&format!("node{}", i), Position::new(x, y)))?;
    }

    graph
}

Benchmark Suite

fn run_performance_benchmark() {
    let test_sizes = vec![100, 250, 500, 750, 1000, 1500];

    for &size in &test_sizes {
        let graph = generate_large_graph(size);

        let start_time = instant::now();
        render_graph(&graph)?;
        let render_time = instant::now() - start_time;

        log_performance_result(size, render_time);
    }
}
