💻 Implementation

How background customization is implemented in Flow-RS

Background Configuration

let background_config = BackgroundConfig {
    color: "#f8fafc".to_string(),        // Background fill color
    pattern_color: "#e2e8f0".to_string(), // Pattern element color
    variant: BackgroundVariant::Dots,    // Pattern type
    size: 20.0,                          // Pattern spacing/size
    opacity: 0.5,                        // Pattern transparency
};

// Render the background
renderer.render_background(&background_config, &viewport)?;

Background Variants

enum BackgroundVariant {
    Dots,      // Circular dots in grid pattern
    Lines,     // Horizontal and vertical lines
    Crosshatch // Diagonal crossing lines
}

// Easy to extend with custom patterns
impl BackgroundVariant {
    fn render_pattern(&self, renderer: &mut Canvas2DRenderer, config: &BackgroundConfig) {
        match self {
            BackgroundVariant::Dots => self.render_dots(renderer, config),
            BackgroundVariant::Lines => self.render_lines(renderer, config),
            BackgroundVariant::Crosshatch => self.render_crosshatch(renderer, config),
        }
    }
}
