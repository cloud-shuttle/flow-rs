Custom Edge Data Structure

#[derive(Clone, Debug)]
struct StyledEdgeData {
    pub edge_type: EdgeType,      // Straight, Bezier, Step, SmoothStep
    pub color: String,            // Hex color codes
    pub thickness: f64,           // Line width
    pub dash_pattern: Option>, // Dash pattern array
    pub label: Option,    // Edge label text
    pub animated: bool,           // Animation flag
}

// Edge type enum
enum EdgeType {
    Straight,
    Bezier,
    Step,
    SmoothStep,
}

Creating Styled Edges

// Create a thick, animated edge
let animated_edge = Edge {
    id: "animated".to_string(),
    source: "process".to_string(),
    target: "output".to_string(),
    data: StyledEdgeData::new(
        EdgeType::Bezier,     // Curved routing
        "#e53e3e",            // Red color
        3.0,                  // 3px thickness
        Some("Animated"),     // Label
        true,                 // Animated
    ),
};

// Create a dashed conditional edge
let dashed_edge = StyledEdgeData::new(
    EdgeType::Bezier,
    "#f56565",
    2.0,
    Some("Dashed"),
    false,
)
.with_dash_pattern(vec![10.0, 5.0]); // 10px dash, 5px gap
