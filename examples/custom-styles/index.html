How custom styling is implemented in Flow-RS

Custom Node Data Structure

// Define custom data with styling properties
#[derive(Clone, Debug)]
struct StyledNodeData {
    pub label: String,
    pub color: String,
    pub shape: NodeShape,    // Rectangle, Rounded, Circle
    pub size: NodeSize,      // Small, Medium, Large
}

// Create styled nodes
let process_node = Node {
    id: "process".to_string(),
    position: Position::new(350.0, 100.0),
    data: StyledNodeData::new(
        "Process",
        "#48bb78",           // Green color
        NodeShape::Rectangle, // Rectangle shape
        NodeSize::Large,      // Large size
    ),
    ..Default::default()
};
